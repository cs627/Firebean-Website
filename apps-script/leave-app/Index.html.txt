/* ============================================================
   回程監看器（build-2026-11-03y）
   背景：實測喺部分瀏覽器／部分情況下，google.script.run 嘅回程
        會靜靜咁消失（伺服器已執行完，前端永遠收唔到）。
        結果：畫面卡住，用戶以為按咗冇反應。
   做法：包住 google.script.run，任何 server 呼叫 12 秒冇回音 →
        上報 Diag + 自動重載一次（重載後由伺服器內嵌資料顯示「真實結果」，
        所以就算回程死咗，用戶都一定睇得到結果）。
   ============================================================ */
(function () {
  try {
    var armed = null, pendingFn = '';
    function disarm() { if (armed) { clearTimeout(armed); armed = null; } }
    function reloadOnce() {
      try {
        var last = +(sessionStorage.getItem('lv_reload') || 0);
        if (Date.now() - last > 20000) {
          sessionStorage.setItem('lv_reload', String(Date.now()));
          location.reload();
        }
      } catch (e) {}
    }
    function armWatch(fnName) {
      disarm(); pendingFn = String(fnName);
      armed = setTimeout(function () {
        armed = null;
        reportErr('nocb:' + pendingFn);
        reloadOnce();
      }, 12000);
    }
    function wrap(o) {
      return new Proxy(o, {
        get: function (t, k) {
          var v = t[k];
          if (typeof v !== 'function') return v;
          return function () {
            var a = arguments;
            if (k === 'withSuccessHandler') {
              var fn = a[0];
              return wrap(v.call(t, function (r) { disarm(); return fn ? fn(r) : undefined; }));
            }
            if (k === 'withFailureHandler') {
              var fe = a[0];
              return wrap(v.call(t, function (e) { disarm(); return fe ? fe(e) : undefined; }));
            }
            armWatch(k);
            return v.apply(t, a);
          };
        }
      });
    }
    var tries = 0;
    function tryInstall() {
      tries++;
      try {
        var sc = window.google && window.google.script;
        if (!sc || !sc.run) {
          if (tries < 24) { setTimeout(tryInstall, 500); return; }
          reportErr('watchdog: no google.script.run after 12s'); return;
        }
        var desc = Object.getOwnPropertyDescriptor(sc, 'run');
        Object.defineProperty(sc, 'run', {
          configurable: true,
          get: function () {
            var base = desc && desc.get ? desc.get.call(sc) : (desc ? desc.value : null);
            return base ? wrap(base) : base;
          }
        });
        if (tries > 1) reportErr('watchdog: on (tries=' + tries + ')');
      } catch (e) {
        try { reportErr('watchdog install EXC: ' + ((e && e.message) || e)); } catch (e2) {}
      }
    }
    tryInstall();
  } catch (e) {
    try { reportErr('watchdog install EXC: ' + ((e && e.message) || e)); } catch (e2) {}
  }
})();

/* ── 共用元件：餘額卡（含公式）────────────────────────────── */
function calcRow(label, expr, result, unit){
  return '<div class="calcRow"><span class="cl">' + esc(label) + '</span>' +
    '<span class="cm">' + esc(expr) + '</span>' +
    '<span class="cr">' + num(result) + (unit ? '<small>' + esc(unit) + '</small>' : '') + '</span></div>';
}
function fmtMonth(iso){
  if (!iso) return '';
  var d = new Date(String(iso).slice(0,10) + 'T00:00:00');
  if (isNaN(d.getTime())) return String(iso);
  var M = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
  return M[d.getMonth()] + ' ' + d.getFullYear();
}
function calcBlockHtml(b, segs){
  // 累積算式：逐段列出（年度中調整 entitlement 時會有多過一段）
  var expr;
  var shown = (segs || []).filter(function(x){ return x.upfront || Number(x.ent) > 0; });
  if (shown.length) {
    expr = shown.map(function(x){
      if (x.upfront) return num(x.ent) + ' days (full year)';
      return '(' + num(x.ent) + ' days \u00d7 ' + x.months + ' months) \u00f7 12 months';
    }).join('   +   ');
  } else {
    expr = '\u2013';
  }
  var used = num(b.used), pend = num(b.pending);
  var carry = Number(b.carryIn || 0);
  var raw = (b.accruedRaw === undefined || b.accruedRaw === null) ? b.accrued : b.accruedRaw;
  var toLbl = fmtMonth(b.accruedTo);
  var rows = calcRow('Accrued' + (toLbl ? ' (to ' + toLbl + ')' : ''), expr, raw, 'days');
  // 有零碎日數先顯示對齊步驟，免得無謂多一行
  if (Number(raw) !== Number(b.accrued)) {
    rows += '<div class="calcRow"><span class="cl">Rounded up to next half day</span>' +
      '<span class="cm">' + num(raw) + ' days \u2192</span>' +
      '<span class="cr">' + num(b.accrued) + '<small>days</small></span></div>';
  }
  if (carry) {
    var cd = (b.carryDetail || []).filter(function(x){ return Number(x.closing) < 0; });
    var cy = cd.length ? cd[cd.length - 1].year : '';
    var ctxt = cy ? 'Over-used in ' + cy + ' \u2014 carried forward' : 'Carried forward from last year';
    rows += calcRow('Carry-over from last year', ctxt, carry, 'days');
  }
  return '<div class="calc">' +
    '<div class="ch">How this is calculated</div>' +
    rows +
    calcRow('Annual leave left',
      'Accrued ' + num(b.accrued) + ' days' +
      (carry ? (carry < 0 ? ' \u2212 ' + num(Math.abs(carry)) : ' + ' + num(carry)) + ' days carry-over' : '') +
      ' \u2212 Used ' + used + ' days \u2212 Pending ' + pend + ' days',
      b.available, 'days') +
  '</div>';
}
var _MON = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
function isoLabel(iso){
  var p = String(iso || '').split('-');
  if (p.length !== 3) return String(iso || '');
  return Number(p[2]) + ' ' + (_MON[Number(p[1]) - 1] || p[1]) + ' ' + p[0];
}
/* 帶入年假提示：正數有期限、負數係超用、過期即失效 */
function carryHtml(b){
  var raw = Number(b.carryRaw || 0);
  if (!raw) return '';
  var last = b.carryExpiry ? isoLabel(b.carryExpiry) : '';
  if (raw < 0) {
    return '<div class="d" style="margin-top:10px;color:#b42318">⚠️ ' + num(Math.abs(raw)) +
      ' day(s) over-used last year, carried forward and already deducted from this year.</div>';
  }
  if (b.carryExpired) {
    var cr = Number(b.carryCredit || 0);
    var extra = cr
      ? ' <b>' + num(cr) + ' day(s)</b> of them were taken before the deadline, so they are not charged against this year\u2019s accrual.'
      : '';
    return '<div class="d" style="margin-top:10px;color:#b45309">⚠️ ' + num(raw) +
      ' day(s) were carried forward from last year, but the deadline passed on <b>' + last +
      '</b> \u2014 they have expired and can no longer be used.' + extra + '</div>';
  }
  return '<div class="d" style="margin-top:10px;color:#12864a">↪️ <b>' + num(raw) +
    ' day(s)</b> carried forward from last year — usable until <b>' + last +
    '</b>, after which any unused days expire.</div>';
}

var ZH_MON = {Jan:'1',Feb:'2',Mar:'3',Apr:'4',May:'5',Jun:'6',
               Jul:'7',Aug:'8',Sep:'9',Oct:'10',Nov:'11',Dec:'12'};
function zhDate(s){
  return s.replace(/\b(Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)\b (\d{4})/g,
                   function(_, m, y){ return y + ' \u5e74 ' + ZH_MON[m] + ' \u6708'; })
          .replace(/\b(Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec)\b (\d{1,2})\b/g,
                   function(_, m, d){ return ZH_MON[m] + ' \u6708 ' + d + ' \u65e5'; });
}
function zhFor(t){
  if (Object.prototype.hasOwnProperty.call(ZH, t)) return zhDate(ZH[t]);
  for (var i = 0; i < ZH_RULES.length; i++) {
    if (ZH_RULES[i][0].test(t)) return zhDate(t.replace(ZH_RULES[i][0], ZH_RULES[i][1]));
  }
  return zhDate(t);
}

function i18nApply(){
  var root = $('root'); if (!root) return;
  var zh = (LANG === 'zh');
  var nodes = [], w = document.createTreeWalker(root, NodeFilter.SHOW_TEXT, null, false), n;
  while ((n = w.nextNode())) nodes.push(n);
  for (var i = 0; i < nodes.length; i++) {
    var nd = nodes[i];
    if (nd.__en === undefined) nd.__en = nd.nodeValue;
    var raw = nd.__en, t = raw.trim();
    if (!t) continue;
    nd.nodeValue = zh ? raw.replace(t, zhFor(t)) : raw;
  }
  var phs = root.querySelectorAll('[placeholder]');
  for (var j = 0; j < phs.length; j++) {
    var el = phs[j];
    if (el.__enph === undefined) el.__enph = el.placeholder;
    el.placeholder = zh ? zhFor(el.__enph) : el.__enph;
  }
  document.documentElement.lang = zh ? 'zh-Hant' : 'en';
  var lb = $('langBtn');
  if (lb) lb.textContent = zh ? 'EN' : '\u4e2d\u6587';
}

}

function reqHtml(r, showButtons){

  var got = String(r.cert || '').trim();
  var h = '<div class="item">' +
    '<div class="top">' +

      '<div class="tag ' + esc(r.status) + '">' + cnStatus(r.status) + '</div>' +
    '</div>' +
    '<div class="d mono">' + fmtDate(r.start) + (r.start !== r.end ? ' \u2192 ' + fmtDate(r.end) : '') +
      ' \u00b7 ' + num(r.days) + ' d' + (half ? ' \u00b7 half day' : '') + '</div>';
  if (r.reason) h += '<div class="d">' + esc(r.reason) + '</div>';
  if (r.eventName) h += '<div class="d">' + esc(r.eventName) +
    (r.otDate ? ' \u00b7 OT ' + fmtDate(r.otDate) : '') + '</div>';
  if (r.type === 'Sick Leave') {
    h += '<div class="d" style="margin-top:8px"><span class="chip ' + (got ? 'ok' : 'miss') + '">' +
         (got ? L('Medical certificate: received') : L('Medical certificate: not received')) + '</span></div>';
  }

  if (r.calId) h += '<div class="d" style="color:#12864a">\u2713 Added to team calendar</div>';

  var isMgr = !!(S.sess && S.sess.isManager);
  if (showButtons && r.status === 'Pending') {
    h += '<div class="row2" style="margin-top:12px">' +
      '<button class="btn sm ok" onclick="decide(\'' + esc(r.id) + '\',\'Approved\')">Approve</button>' +
      '<button class="btn sm bad" onclick="decide(\'' + esc(r.id) + '\',\'Rejected\')">Reject</button>' +
    '</div>';
    if (isMgr && r.type === 'Sick Leave') {
      h += '<div class="row2" style="margin-top:8px">' +
        '<button class="btn sm' + (got ? '' : ' ok') + '" onclick="markCert(\'' + esc(r.id) + '\',\'' + (got ? '' : 'RECEIVED') + '\')">' +

  var todayIso = (S.sess && S.sess.today) || '';
  var notStarted = !r.start || !todayIso || String(r.start) >= todayIso;
  if (r.status === 'Approved' && !notStarted)
    h += '<div class="d" style="color:#8b93a3">\u2014 already started, cannot be cancelled</div>';

      '<button class="btn sm bad" onclick="cancelReq(\'' + esc(r.id) + '\',\'' + esc(r.status) + '\')">' + lbl + '</button>' +
    '</div>';
  }
  return h + '</div>';
}

function onDurChange(){
  if ($('f-dur').value.indexOf('Half') >= 0) {
    $('f-end').value = $('f-start').value;
    $('f-end').disabled = true;
  } else {

}
function onDateChange(){
  if ($('f-dur').value.indexOf('Half') >= 0) $('f-end').value = $('f-start').value;
  if (!$('f-end').value || $('f-end').value < $('f-start').value) $('f-end').value = $('f-start').value;
  updatePreview();
}
function updatePreview(){
  var p = $('f-preview'); if (!p) return;
  if (p.__enHtml !== undefined) { try { delete p.__enHtml; } catch (e) { p.__enHtml = undefined; } }
  var s = $('f-start').value, e = $('f-end').value, d = $('f-dur').value;
  if (!s || !e || e < s) { p.innerHTML = ''; return; }
  var n = workdays(s, e);
  if (d.indexOf('Half') >= 0) n = 0.5;
  var type = $('f-type').value;
  var txt = 'This request: <b>' + num(n) + ' day' + (n === 1 ? '' : 's') + '</b> (weekends &amp; public holidays excluded)';
  if (type === 'Annual Leave' && S.data && S.data.balance) {
    var left = Number(S.data.balance.available) - n;
    txt += '<br>Annual leave left after booking: <b style="color:' +
      (left < 0 ? '#ff3333' : '#12864a') + '">' + num(left) + ' days</b>';
    if (left < 0) txt += ' \u2014 exceeds your balance';
    txt += '<br><span style="color:#8b93a3">Contract entitlement ' + num(S.data.balance.entitlement) +
      ' days/year (full 12 months) \u00b7 available to book now: ' + num(S.data.balance.available) + ' days</span>';
  }
  p.innerHTML = txt;
  if (typeof i18nApply === 'function') i18nApply();
}
function workdays(s, e){
  var hol = (S.data && S.data.holidays) ? S.data.holidays : [];
  var d = new Date(s + 'T00:00:00'), end = new Date(e + 'T00:00:00'), n = 0;
  while (d <= end) {
    var dow = d.getDay();
    var iso = d.getFullYear() + '-' + pad(d.getMonth()+1) + '-' + pad(d.getDate());
    if (dow !== 0 && dow !== 6 && hol.indexOf(iso) < 0) n++;
    d.setDate(d.getDate() + 1);
  }
  return n;
}
function pad(n){ return n < 10 ? '0' + n : '' + n; }

    eventName: $('ot-box').style.display !== 'none' ? $('f-event').value : ''
  };
  if (!payload.startDate || !payload.endDate) { toast('Please pick your dates', true); return; }

  btn.disabled = true; btn.textContent = L('Submitting\u2026');
  google.script.run.withSuccessHandler(function(r){
    btn.disabled = false; btn.textContent = L('Submit Request');
    if (r && r.ok){
      toast(r.message || 'Request submitted');
      $('f-reason').value = '';
      if ($('f-ot')) $('f-ot').value = '';
      if ($('f-event')) $('f-event').value = '';
      loadDashboard();
    } else {
      toast((r && r.error) || 'Could not submit', true);
    }
  }).withFailureHandler(function(e){
    btn.disabled = false; btn.textContent = L('Submit Request');
    toast(e && e.message ? e.message : 'Could not submit', true);
  }).submitLeave(payload, WHO);
}

function markCert(id, value){

  google.script.run.withSuccessHandler(function(r){
    if (r && r.ok){ toast(value ? L('Marked as received') : L('Marked as not received')); loadManager(); }
    else toast((r && r.error) || 'Could not update', true);
  }).withFailureHandler(function(e){
    toast(e && e.message ? e.message : 'Could not update', true);
  }).markCertificate(id, value, WHO);
}

function loadManager(){
  google.script.run.withSuccessHandler(function(m){
    if (!m || !m.ok) return;
    S.mgr = m;
    renderManager();
  }).withFailureHandler(function(e){ console.error(e); }).getManagerView(WHO);
}

function renderManager(){
  var m = S.mgr; if (!m) return;
  var el;
  ['mgr-pending','mgr-people','mgr-recent'].forEach(function(id){
    var e = $(id); if (e) { try { delete e.__enHtml; } catch (x) { e.__enHtml = undefined; } }
  });

  el = $('mgr-pending');
  if (el) {
    el.innerHTML = m.pending.length
      ? m.pending.map(function(r){ return reqHtml(r, true); }).join('')
      : '<div class="empty">Nothing waiting</div>';
  }
  el = $('mgr-people');
  if (el) {
    el.innerHTML = m.people.map(function(p){
      var b = p.balance;
      var neg = Number(b.available) < 0;
      return '<div class="item pRow" onclick="openMember(\'' + esc(p.email) + '\')">' +
        '<div class="top" style="align-items:center">' +
          '<div style="display:flex;align-items:center;gap:14px;min-width:0">' +
            avatarHtml(p.avatar, p.name, 54) +
            '<div style="min-width:0">' +
              '<div class="pname">' + esc(p.name) + '</div>' +
              '<div class="prole">' + esc(p.role) + '</div>' +
            '</div>' +
          '</div>' +
          '<div style="display:flex;align-items:center;gap:10px;flex:0 0 auto">' +
            '<div class="pbal" style="color:' + (neg ? '#ff3333' : '#0a0a0a') + '">' +
              num(b.available) + '<small>days</small></div>' +
            '<span class="chev">\u203a</span>' +
          '</div>' +
        '</div>' +
        '<div class="d mono" style="margin-top:11px">Ent ' + num(b.entitlement) +
        ' \u00b7 Accr ' + num(b.accrued) + ' \u00b7 Used ' + num(b.used) +
        ' \u00b7 Pend ' + num(b.pending) + '</div></div>';
    }).join('');
  }
  el = $('mgr-recent');
  if (el) {
    el.innerHTML = m.recent.length
      ? m.recent.slice(0, 15).map(function(r){ return reqHtml(r, false); }).join('')
      : '<div class="empty">No records</div>';
  }
  if (typeof iconsApply === 'function') iconsApply();
  if (typeof i18nApply === 'function') i18nApply();
}

  tab('v-member');
  var el = $('v-member');

function decide(id, decision){
  if (!window.confirm(L(decision === 'Approved' ? 'Approve this request?' : 'Reject this request?'))) return;
  var comment = window.prompt(
    decision === 'Approved'
      ? L('Note (optional \u2014 saved to the record and emailed)')
      : L('Reason for rejection (emailed to the staff member)'),
    ''
  );
  if (comment === null) return;

  toast('Processing\u2026');
  google.script.run.withSuccessHandler(function(r){
    if (r && r.ok){ toast(r.message || 'Done'); loadDashboard(); }
    else { toast((r && r.error) || 'Could not process', true); }
  }).withFailureHandler(function(e){
    toast(e && e.message ? e.message : 'Could not process', true);
  }).decideLeave(id, decision, comment, WHO);
}

function cancelReq(id, status){
  var msg = L((status === 'Approved')
    ? 'Cancel this APPROVED leave?\n\nThe days will be returned to the balance and the calendar entry will be removed.'
    : 'Cancel this request?');
  if (!window.confirm(msg)) return;
  var note = window.prompt(L('Reason for cancellation (optional \u2014 emailed to the staff member)'), '');
  if (note === null) return;

  toast('Cancelling\u2026');
  google.script.run.withSuccessHandler(function(r){
    if (r && r.ok){
      toast('Cancelled' + (r.calRemoved ? ' \u2014 calendar entry removed' : '') + (r.note || ''), false);
      loadDashboard();
    } else {
      toast((r && r.error) || 'Could not cancel', true);
    }
  }).withFailureHandler(function(e){
    toast(e && e.message ? e.message : 'Could not cancel', true);
  }).cancelLeave(id, note, WHO);
}
